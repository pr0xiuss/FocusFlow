        category=models.CharField(max_length=50,blank=True,null=True)
        completed=models.BooleanField(default=False)
        created_at=models.DateTimeField(auto_now_add=True)
        
        parent_task=models.ForeignKey('self',on_delete=models.CASCADE,null=True,blank=True,related_name='subtasks')
    #   another foriegn key points to Task model creating recursive relationship creating subtasks

        def __str__(self):    --> magic method __ defines string representation of object
            return self.title
        
        def get_depth(self):
            depth=0
            cur=self
            while cur.parent_task:
                depth+=1
                cur=cur.parent_task
            return depth

step 12: define urls for path
    URLs files define how incomming HTTP requests are routed to views (which view handles which URL)
    
    from django.urls import path
    from .views import TaskListCreateView,TaskRetrieveUpdateDestroyView

    urlpatterns=[
        path('tasks/',TaskListCreateView.as_view(),name='task-list-create'),
        path('tasks/<int:pk>/',TaskRetrieveUpdateDestroyView.as_view(),name='task-detail')
    ]

    when user sends API at tasks/ this rule is matched so jango runs TaskListCreateView
    .as_view() turns this class into callable view that jango can use
    if user sends get request --> returns list of all tasks
    if user sends post request --> creates new task

    if goes to tasks/<int:pk>/ it captures number(say 5) as pk jango sends this to TaskRetrieveUpdateDestroyView
    GET → show details of task with id=5.
    PUT/PATCH → update task with id=5.
    DELETE → delete task with id=5.

    URL matches → Django calls the view → view uses serializer → data from/to database → JSON response returned to user.

step 16: Create a User Registration API Endpoint
    from django.contrib.auth.models import User
    class UserSerializer(serializers.ModelSerializer):
        pwd=serializers.CharField(write_only=True,required=True,style={'input_type':'password'})
        pwd2=serializers.CharField(write_only=True,required=True,style={'input_type':'password'})

        class Meta:
            model=User
            fields={'username','email','pwd','pwd2'}
            extra_kwargs={'pwd':{'write_only':True}}
            
        def validate(self,data):
            if data['pwd']!=data['pwd2']:
                raise serializers.ValidationError({"pwd":"Password fields didn't match!"})
            return data
        
        def create(self,validated_data):
            validated_data.pop('pw2')
            user=User.objects.create_user(
                username=validated_data['username'],
                email=validated_data.get('email',''),
                password=validated_data['password']
            )
            return user
    
    write_only-->accepted on input but not returned in responses
    field --> lists which fields the serializer exposes/accepts
    extra_kwargs --> provides extra arguments for auto-generated fields

step 19: add filter and search
    def get_queryset(self):
            queryset=Task.objects.filter(user=self.request.user, parent_task__isnull=True).order_by('id')
            completed_param=self.request.query_params.get('completed',None)
            category_param=self.request.query_params.get('category',None)
            search_param=self.request.query_params.get('search',None)

            #filter by completed
            if completed_param is not None:
                val=str(completed_param).strip().lower()
                is_completed=val in ('true', '1', 'yes')
                queryset=queryset.filter(completed=is_completed)

            #filter by category
            if category_param:
                queryset=queryset.filter(category__iexact=category_param)

            #search
            if search_param:
                queryset=queryset.filter(
                    Q(title__icontains=search_param) | Q(description__icontains=search_param)
                )
            
            return queryset.order_by('id')

            icontains --> case insensitive
            filters by:
                completed=true/false , category=Work , search=keyword
